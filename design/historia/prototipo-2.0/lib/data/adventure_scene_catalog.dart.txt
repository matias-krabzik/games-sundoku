class AdventureSceneCatalog {
  AdventureSceneCatalog._();

  static final List<AdventureScene> scenes = List.unmodifiable([
    for (var level = 1; level <= _valley.length; level++)
      for (var round = 0; round < 3; round++)
        AdventureScene(
          id: '${mapLevelId(level)}/after-round-${round + 1}',
          levelNumber: level,
          roundIndex: round,
          title: kValleyLevelNames[level - 1],
          dialogues: List.unmodifiable(_valley[level - 1][round]),
        ),
  ]);

  static AdventureScene? byId(String id) =>
      scenes.where((scene) => scene.id == id).firstOrNull;

  static AdventureScene? afterPuzzle(String puzzleId) => scenes
      .where(
        (scene) =>
            '${mapLevelId(scene.levelNumber)}/sudoku-${scene.roundIndex + 1}' ==
            puzzleId,
      )
      .firstOrNull;
